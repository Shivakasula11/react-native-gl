# Profile Registration
# The following Trigger inserts record into profiles table during signup

create function public.handle_new_user() 
returns trigger 
as $$
begin
    insert into public.profiles (user_id, first_name, last_name, email)
    values (
        new.id, -- Changed from new.user_id to new.id
        coalesce(new.raw_user_meta_data->>'first_name', ''), -- Ensure it's not NULL
        coalesce(new.raw_user_meta_data->>'last_name', ''),
        email
    );
    return new;
exception when others then
    raise warning 'Failed to insert profile for user_id: %', new.id;
    return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
after insert on auth.users 
for each row
execute procedure public.handle_new_user();
